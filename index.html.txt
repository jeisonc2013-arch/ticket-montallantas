            document.getElementById('inputFecha').value = today;

            // Generate initial ticket number
            document.getElementById('ticketNumBadge').innerText = `#TK-${ticketCounter}`;
            document.getElementById('ticketNumberDisplay').innerText = `TK-${ticketCounter}`;

            // Add default initial service for demonstration
            serviceItems.push({
                id: Date.now(),
                nombre: "Montaje de Llantas",
                unidades: 2,
                precio: 15000,
                total: 30000
            });
            serviceItems.push({
                id: Date.now() + 1,
                nombre: "Balanceo Digital",
                unidades: 2,
                precio: 18000,
                total: 36000
            });

            // Bind Event Listeners for real-time updates
            document.getElementById('inputFecha').addEventListener('input', updateTicketPreview);
            document.getElementById('inputCiudad').addEventListener('input', updateTicketPreview);
            document.getElementById('inputCliente').addEventListener('input', updateTicketPreview);
            document.getElementById('inputPlaca').addEventListener('input', updateTicketPreview);
            document.getElementById('inputKm').addEventListener('input', updateTicketPreview);
            document.getElementById('inputObservaciones').addEventListener('input', updateTicketPreview);
            document.getElementById('inputTecnico').addEventListener('input', updateTicketPreview);
            document.getElementById('inputGarantia').addEventListener('change', updateTicketPreview);

            renderServices();
            updateTicketPreview();
        });

        // Apply Preset selection to custom service input & price
        function applyPresetService() {
            const select = document.getElementById('selectPresetService');
            const selectedOption = select.options[select.selectedIndex];
            const customInput = document.getElementById('inputCustomService');
            const priceInput = document.getElementById('inputPrecio');

            if (select.value) {
                customInput.value = select.value;
                const defaultPrice = selectedOption.getAttribute('data-precio') || 0;
                priceInput.value = defaultPrice;
            }
        }

        // Add a new service item to list
        function addServiceItem() {
            const selectPreset = document.getElementById('selectPresetService').value;
            const customName = document.getElementById('inputCustomService').value.trim();
            const unidades = parseInt(document.getElementById('inputUnidades').value) || 1;
            const precio = parseFloat(document.getElementById('inputPrecio').value) || 0;

            const serviceName = customName || selectPreset;

            if (!serviceName) {
                showToast("Por favor ingresa o selecciona un tipo de servicio.");
                return;
            }

            const newItem = {
                id: Date.now(),
                nombre: serviceName,
                unidades: unidades,
                precio: precio,
                total: unidades * precio
            };

            serviceItems.push(newItem);

            // Reset Item Form Fields
            document.getElementById('selectPresetService').value = "";
            document.getElementById('inputCustomService').value = "";
            document.getElementById('inputUnidades').value = "1";
            document.getElementById('inputPrecio').value = "";

            renderServices();
            updateTicketPreview();
            showToast("Servicio agregado al ticket");
        }

        // Remove a service item
        function removeServiceItem(id) {
            serviceItems = serviceItems.filter(item => item.id !== id);
            renderServices();
            updateTicketPreview();
            showToast("Servicio eliminado");
        }

        // Render Service Items in Form & Ticket Preview
        function renderServices() {
            const formTableBody = document.getElementById('servicesFormTableBody');
            const emptyMsg = document.getElementById('emptyServicesMessage');
            const ticketTableBody = document.getElementById('ticketServicesList');

            // Clear table bodies
            formTableBody.innerHTML = '';
            ticketTableBody.innerHTML = '';

            if (serviceItems.length === 0) {
                emptyMsg.classList.remove('hidden');
            } else {
                emptyMsg.classList.add('hidden');
            }

            serviceItems.forEach((item, index) => {
                // Render Form Table Row
                const formRow = document.createElement('tr');
                formRow.className = "border-b hover:bg-gray-50";
                formRow.innerHTML = `
                    <td class="p-2.5 font-medium">${item.nombre}</td>
                    <td class="p-2.5 text-center font-bold">${item.unidades}</td>
                    <td class="p-2.5 text-right">${formatMoney(item.precio)}</td>
                    <td class="p-2.5 text-right font-bold text-[#221f1f]">${formatMoney(item.total)}</td>
                    <td class="p-2.5 text-center">
                        <button onclick="removeServiceItem(${item.id})" class="text-red-500 hover:text-red-700 font-bold px-1.5 py-0.5 rounded">
                            <i class="fa-solid fa-trash-can"></i>
                        </button>
                    </td>
                `;
                formTableBody.appendChild(formRow);

                // Render Ticket Preview Row
                const ticketRow = document.createElement('tr');
                ticketRow.className = "border-b border-gray-100";
                ticketRow.innerHTML = `
                    <td class="py-2.5 pr-2 font-medium">${item.nombre}</td>
                    <td class="py-2.5 px-2 text-center font-semibold">${item.unidades}</td>
                    <td class="py-2.5 px-2 text-right">${formatMoney(item.precio)}</td>
                    <td class="py-2.5 pl-2 text-right font-bold">${formatMoney(item.total)}</td>
                `;
                ticketTableBody.appendChild(ticketRow);
            });
        }

        // Update live ticket display content
        function updateTicketPreview() {
            const fechaVal = document.getElementById('inputFecha').value;
            const ciudadVal = document.getElementById('inputCiudad').value || 'Bogotá D.C.';
            const clienteVal = document.getElementById('inputCliente').value || 'Cliente General';
            const placaVal = document.getElementById('inputPlaca').value || '---';
            const kmVal = document.getElementById('inputKm').value;
            const obsVal = document.getElementById('inputObservaciones').value;
            const tecnicoVal = document.getElementById('inputTecnico').value || 'Técnico Responsable';
            const garantiaVal = document.getElementById('inputGarantia').value;

            // Formatted Date
            if (fechaVal) {
                const parts = fechaVal.split('-');
                document.getElementById('ticketDateDisplay').innerText = `${parts[2]}/${parts[1]}/${parts[0]}`;
            }

            document.getElementById('ticketCityDisplay').innerText = ciudadVal;
            document.getElementById('ticketClientDisplay').innerText = clienteVal;
            document.getElementById('ticketPlateDisplay').innerText = placaVal.toUpperCase();
            document.getElementById('ticketKmDisplay').innerText = kmVal ? `${Number(kmVal).toLocaleString('es-CO')} Km` : 'N/A';
            
            document.getElementById('ticketObsDisplay').innerText = obsVal.trim() !== "" ? obsVal : 'Sin observaciones registradas para este servicio.';
            document.getElementById('ticketTechDisplay').innerText = tecnicoVal;
            document.getElementById('ticketGarantiaDisplay').innerText = garantiaVal;

            // Calculate Totals
            const grandTotal = serviceItems.reduce((acc, curr) => acc + curr.total, 0);
            document.getElementById('ticketSubtotalDisplay').innerText = formatMoney(grandTotal);
            document.getElementById('ticketTotalDisplay').innerText = formatMoney(grandTotal);
        }

        // Reset form for a new ticket
        function resetForm() {
            ticketCounter = Math.floor(1000 + Math.random() * 9000);
            document.getElementById('ticketNumBadge').innerText = `#TK-${ticketCounter}`;
            document.getElementById('ticketNumberDisplay').innerText = `TK-${ticketCounter}`;

            document.getElementById('inputCliente').value = '';
            document.getElementById('inputPlaca').value = '';
            document.getElementById('inputKm').value = '';
            document.getElementById('inputObservaciones').value = '';
            document.getElementById('inputTecnico').value = '';

            serviceItems = [];
            renderServices();
            updateTicketPreview();
            showToast("Formulario limpiado para nuevo ticket");
        }

        // Notification Helper
        function showToast(message) {
            const toast = document.getElementById('toastNotification');
            const toastText = document.getElementById('toastText');
            toastText.innerText = message;
            
            toast.classList.remove('translate-y-20', 'opacity-0');
            toast.classList.add('translate-y-0', 'opacity-100');

            setTimeout(() => {
                toast.classList.remove('translate-y-0', 'opacity-100');
                toast.classList.add('translate-y-20', 'opacity-0');
            }, 3000);
        }
    </script>
